def main() -> int:
    try:
        sys.stdout.reconfigure(encoding="utf-8", errors="replace")
    except Exception:
        pass
    parser = argparse.ArgumentParser()
    parser.add_argument("--repo", type=Path, required=True)
    parser.add_argument("mode", choices=("correct", "begin-replay", "finish"))
    parser.add_argument("--id", type=int)
    parser.add_argument("--verdict-file", type=Path)
    parser.add_argument("--evidence")
    parser.add_argument("--evidence-file", type=Path, help="a file holding the evidence text (prose never passes the shell)")
    parser.add_argument("--trees-json", type=Path)
    args = parser.parse_args()
    sys.path.insert(0, str(args.repo.resolve()))
    from ai_player.gates import ledger
    from ai_player.gates.provenance import line_digest, provenance

    print(f"ledger file: {ledger.LEDGER}")
    if args.mode == "finish":
        verdict = " ".join(args.verdict_file.read_text(encoding="utf-8").split())
        evidence = (" ".join(args.evidence_file.read_text(encoding="utf-8").split()) if args.evidence_file is not None
                    else args.evidence)
        ledger.finish(args.id, verdict, evidence)
        print(f"evidence: {evidence}")
        print(f"finished entry #{args.id}: {verdict}")
        return 0
    if args.mode == "begin-replay":
        trees = json.loads(args.trees_json.read_text(encoding="utf-8"))
        numbers = replay_numbers(trees)
        bounds = ledger.numbers_sha256(numbers)
        code = provenance(args.repo.resolve())
        print("numbers " + json.dumps(numbers, ensure_ascii=False, sort_keys=True))
        print(f"numbers sha256 {bounds}")
        print("code: " + code)
        entry, lines = ledger.begin("story", "r5_fix_replay", "replay", 5301, 5301, bounds, line_digest(code),
                                    note=REPLAY_NOTE)
        print("\n".join(lines))
        print("ENTRY " + json.dumps(entry, ensure_ascii=False))
        return 0
    with ledger._locked(ledger.LEDGER):
        data = ledger.load(ledger.LEDGER)
        entries = data["entries"]
        by_id = {e["id"]: e for e in entries}
        expect = {43: ("r5_probe", 5301, 5304, "53e0c26e7812db47"), 44: ("r5_ab_hookcheck", 5301, 5301, "29a5dc81286162ff"),
                  45: ("r5_ab", 5351, 5354, "29a5dc81286162ff"), 46: ("r5_ab_control", 5355, 5355, "29a5dc81286162ff")}
        for number, (driver, first, last, tree) in expect.items():
            entry = by_id.get(number)
            if entry is None or (entry["driver"], entry["first"], entry["last"], entry["tree"]) != (driver, first, last, tree):
                raise SystemExit(f"REFUSED: entry #{number} is not as the A/B left it: {entry}")
        if 47 in by_id or max(by_id) != 46:
            raise SystemExit(f"REFUSED: the ledger's last entry is #{max(by_id)}, not #46")
        by_id[43].update(verdict=V43, evidence=E43)
        for number in (44, 45, 46):
            by_id[number]["evidence"] = E_AB
        entries.append({"id": 47, "family": "story", "driver": "r5_probe", "purpose": "replay", "first": 5301, "last": 5303,
                        "status": "done", "registered": None, "bounds_sha256": ledger.numbers_sha256(numbers_47()),
                        "tree": "29a5dc81286162ff", "started": "2026-10-28 21:25:50", "ended": "2026-10-28 21:31:05",
                        "verdict": V47, "evidence": E_AB, "note": N47, "replays": [43]})
        problems = ledger.check(data)
        if problems:
            raise SystemExit(f"REFUSED: the corrected ledger would not be well formed: {problems}")
        ledger._save(data, ledger.LEDGER)
    for number in (43, 44, 45, 46, 47):
        print("ENTRY " + json.dumps(next(e for e in entries if e["id"] == number), ensure_ascii=False))
    return 0


if __name__ == "__main__":
    sys.exit(main())
