def reached(row: dict | None) -> tuple[bool, str]:

def files_of(root: Path) -> dict[str, bytes]:
    out = {}
    for path in sorted(root.rglob("*")):
        if not path.is_file():

        rel = path.relative_to(root).as_posix()
        parts = set(rel.split("/"))
