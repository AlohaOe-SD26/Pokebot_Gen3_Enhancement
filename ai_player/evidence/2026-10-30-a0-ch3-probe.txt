"""A0, the AI's FIRST chapter-3 runs (the AI trainer's design, `ai-research-notes/2026-10-30-AI-TRAINER-DESIGN.md`, section 4).

Three launcher runs, in parallel, each in its own disposable profile, from the AI's OWN chapter-2 ends (the 2026-10-30
r1base arms CH2, CH2T and CH2M, each with its world memory and its chain's personality), with the target ch3_rustboro.
DEVELOPMENT seeds 5901-5903 (one ledger entry; never graded). Each run goes through the real launcher (its purity gate per
launch); what a run reached is read from the GAME (the harness's truth) on the state it left, freshness-checked
(L5.THIS_RUN). Writes one evidence file and closes the ledger entry. A probe, not a guard: it has no PASS.

    py -3.12 a0_probe.py            the orchestrator (refuses unless the design note is committed and pushed)
    py -3.12 a0_probe.py --child N  one run (N = 1, 2, 3)
"""
import hashlib
import json
import os
import subprocess
import sys
import time
from pathlib import Path

REPO = Path("C:/Remote Projects/Pokebot_Gen3_Enhancement")
sys.path.insert(0, str(REPO))
HERE = Path(__file__).resolve().parent
NOTE = "PROJECT KNOWLEDGE/ai-research-notes/2026-10-30-AI-TRAINER-DESIGN.md"
RUNS = REPO / "sandbox-pokebot" / "profiles" / "ai-step5-play" / "ai_player_runs"
STARTS = {1: ("20261030-135558-CH2", "Catcher"), 2: ("20261030-135912-CH2T", "Story"), 3: ("20261030-140134-CH2M", "Trainer")}
SEED0 = 5900
FRAMES = 600000
WALL = 3600.0
TARGET = "ch3_rustboro"
PREDICTION = ("A0 prediction (the design's section 4, registered by the commit that filed it): recon gap 1. The run ends on "
              "ROUTE104 at y > 37, with woods state 1 and the Rustboro flag clear. CONTROL: the arrival on ROUTE104 north at "
              "y < 31 would refute it.")


def git(*args: str) -> str:
    return subprocess.run(["git", "-C", str(REPO), *args], capture_output=True, text=True, check=True).stdout.strip()


def child(n: int) -> None:
    from ai_player.gates import live_step5 as L5
    L5.PROFILE = f"ai-ch3-probe-{n}"
    run_name, personality = STARTS[n]
    src = RUNS / run_name
    earlier = []
    for line in (src / "actions.jsonl").read_text(encoding="utf-8").splitlines():
        try:
            earlier.append(json.loads(line))
        except json.JSONDecodeError:
            pass
    arm = L5.Arm(f"A0-{n}", f"chapter 3 from {run_name}'s own end, {personality}", FRAMES, WALL,
                 L5.play_env(personality, None, TARGET), seed=SEED0 + n, fresh=False, state_from="CH2")
    carried = {"CH2": (src / "end_state.ss1", src / "world.jsonl", earlier)}
    run = L5.run_arm(arm, L5.launcher.resolve_build_root(None), carried)
    L5.check_after_is_this_run(run)
    after = run.after or {}
    out = {
        "n": n, "start": run_name, "personality": personality, "seed": SEED0 + n, "exit": run.exit_code,
        "elapsed": round(run.elapsed or 0, 1), "run_dir": str(run.run_dir), "command": run.command,
        "verdict": run.verdict, "checks": [getattr(c, "__dict__", str(c)) for c in run.checks], "notes": run.notes,
        "truth_keys": sorted(after.keys()), "map_before": (run.before or {}).get("map"), "map_after": after.get("map"),
        "marks_before": L5.marks(run.before), "marks_after": L5.marks(after), "stats_after": L5.stats(after),
        "milestones_told": [str(r.get("reason"))[:300] for r in L5.tells(run, "MILESTONE ")],
        "goals_last": [f"f{r.get('frame')} {r.get('action')}: {str(r.get('reason'))[:220]}" for r in run.records
                       if r.get("decision") == "goal"][-40:],
        "log_tail": run.ai_log.splitlines()[-40:],
        "harness_stop": run.measures.get("harness_stop"),
    }
    (HERE / f"a0-out-{n}.json").write_text(json.dumps(out, indent=1, ensure_ascii=False, default=str), encoding="utf-8")
    print(f"[A0-{n}] exit {run.exit_code} elapsed {run.elapsed:.0f}s map {after.get('map')}", flush=True)


def orchestrate() -> int:
    from ai_player.gates import ledger
    from ai_player.gates import live_step5 as L5
    if not git("log", "-1", "--format=%h", "--", NOTE):
        raise SystemExit("REFUSED: the design note (A0's registered prediction) is not committed")
    git("fetch", "-q", "origin")
    if git("rev-parse", "HEAD") != git("rev-parse", "origin/main"):
        raise SystemExit("REFUSED: HEAD is not origin/main: the prediction must be pushed before A0 runs")
    if git("status", "--porcelain"):
        raise SystemExit("REFUSED: the checkout is not clean")
    head = git("rev-parse", "--short", "HEAD")
    tree = git("rev-parse", "HEAD^{tree}")[:16]
    entry, lines = ledger.begin("story", "a0_probe", "development", SEED0 + 1, SEED0 + 3,
                                hashlib.sha256(PREDICTION.encode("utf-8")).hexdigest(), tree,
                                note=f"A0, the AI's first chapter-3 runs (AI trainer design 2026-10-30, section 4), HEAD "
                                     f"{head}; the prediction's sha256 is the bounds field")
    print("\n".join(lines), flush=True)
    env = {**os.environ, "PYTHONUTF8": "1", "PYTHONDONTWRITEBYTECODE": "1"}
    procs = {}
    for n in STARTS:
        log = open(HERE / f"a0-child-{n}.log", "w", encoding="utf-8")
        procs[n] = (subprocess.Popen([sys.executable, "-B", str(Path(__file__).resolve()), "--child", str(n)], cwd=str(REPO),
                                     env=env, stdout=log, stderr=subprocess.STDOUT), log)
        time.sleep(20)
    for n, (proc, log) in procs.items():
        proc.wait()
        log.close()
    results = []
    for n in STARTS:
        path = HERE / f"a0-out-{n}.json"
        results.append(json.loads(path.read_text(encoding="utf-8")) if path.is_file() else {"n": n, "missing": True})
    ev = L5.EVIDENCE_DIR / "2026-10-30-a0-ch3-probe.txt"
    out = [f"A0 - DEVELOPMENT PROBE, NOT GRADED, NO PASS: the AI's FIRST chapter-3 runs (the AI trainer's design, "
           f"`{NOTE}`, section 4)",
           f"date: {time.strftime('%Y-%m-%d %H:%M:%S')}   HEAD {head} (tree {tree})   seeds {SEED0 + 1}-{SEED0 + 3}, "
           f"ledger entry #{entry['id']} (development)",
           f"setup: three launcher runs in parallel, each in its own profile, from the AI's own chapter-2 ends (r1base CH2, CH2T, "
           f"CH2M) with their world memory; target {TARGET}; {FRAMES} frames, {WALL:g} s; the 12-lane bot tuner running",
           f"PREDICTION (registered before the runs): {PREDICTION}",
           "Each run: what the GAME says it reached (marks, read by the harness on the state the run left) and how it ended."]
    for r in results:
        out.append("")
        if r.get("missing"):
            out.append(f"RUN A0-{r['n']}: NO RESULT FILE (see the child log in the session scratch)")
            continue
        out.append(f"RUN A0-{r['n']}: from {r['start']}, {r['personality']}, seed {r['seed']}: exit {r['exit']}, "
                   f"{r['elapsed']} s, run dir {r['run_dir']}")
        out.append(f"  map before {json.dumps(r['map_before'], ensure_ascii=False)}  ->  after "
                   f"{json.dumps(r['map_after'], ensure_ascii=False)}")
        out.append(f"  marks before: {json.dumps(r['marks_before'], ensure_ascii=False)[:1500]}")
        out.append(f"  marks after:  {json.dumps(r['marks_after'], ensure_ascii=False)[:1500]}")
        out.append(f"  freshness and checks: {json.dumps(r['checks'], ensure_ascii=False, default=str)[:1500]}")
        verdict = r.get("verdict") or {}
        out.append(f"  verdict file: {json.dumps(verdict, ensure_ascii=False, default=str)[:1500]}")
        out.append(f"  harness stop: {json.dumps(r.get('harness_stop'), ensure_ascii=False, default=str)[:600]}")
        out.append("  the AI's milestones (told): " + " | ".join(r["milestones_told"])[:1500])
        out.append("  the AI's last goals:")
        out += [f"    {g}" for g in r["goals_last"][-25:]]
        out.append("  the end of the AI's log:")
        out += [f"    {line[:300]}" for line in r["log_tail"][-30:]]
        out += [f"  note: {note}" for note in r["notes"]]
    ev.write_text("\n".join(out) + "\n", encoding="utf-8")
    reached = [r for r in results if not r.get("missing")]
    ledger.finish(entry["id"], f"DEVELOPMENT, not graded: A0, the AI's first chapter-3 runs ({len(reached)} of 3 ran; read "
                               f"the evidence)", str(ev.relative_to(REPO)).replace("\\", "/"))
    print(f"[A0] evidence: {ev}", flush=True)
    for r in reached:
        print(f"[A0-{r['n']}] exit {r['exit']} {r['elapsed']}s map {r['map_after']} marks {json.dumps(r['marks_after'])[:600]}",
              flush=True)
    return 0


if __name__ == "__main__":
    if len(sys.argv) == 3 and sys.argv[1] == "--child":
        child(int(sys.argv[2]))
    else:
        sys.exit(orchestrate())
