def tree_digest(root: Path) -> str:
    h = hashlib.sha256()
    for p in sorted(x for x in root.rglob("*") if x.is_file() and "__pycache__" not in x.parts):
        h.update(p.relative_to(root).as_posix().encode() + b"\0" + hashlib.sha256(p.read_bytes()).digest())
    return h.hexdigest()

def in_tunnel(record: dict) -> bool:
    return "RUSTURF TUNNEL(" in str(record.get("digest") or "")


def analyse(records: list[dict], ai_log: str) -> dict:
    """The facts each batch's prediction names, read from the run's own action log (the AI's records) and its log."""
    out: dict = {}
    out["instructed"] = [(r.get("frame"), r.get("action"), str(r.get("reason"))[:120]) for r in records
                         if r.get("decision") == "instructed"]
    out["unexpected_multichoice"] = ai_log.count("an unexpected multichoice on screen")
    out["call_exit_presses"] = sum(1 for r in records if str(r.get("action", "")).startswith("press")
                                   and "the call has ended" in str(r.get("reason")))
    first = next((r for r in records if isinstance(r.get("observation"), dict)
                  and isinstance(r["observation"].get("player"), dict)), None)  # a New Game's first look has no player
    out["first_look_player_name"] = first["observation"]["player"].get("name") if first else None
    goals = [r for r in records if r.get("decision") == "goal"]
    tunnel_goals = [r for r in goals if in_tunnel(r)]
    out["goals"] = len(goals)
    out["tunnel_decisions"] = len(tunnel_goals)
    out["first_tunnel_frame"] = next((r.get("frame") for r in records if in_tunnel(r)), None)
    out["first_frame"] = next((r.get("frame") for r in records if isinstance(r.get("frame"), int)), None)
    out["tunnel_goal_kinds"] = {}
    for r in tunnel_goals:
        key = str((r.get("extra") or {}).get("key") or r.get("action")).split(":")[0]
        out["tunnel_goal_kinds"][key] = out["tunnel_goal_kinds"].get(key, 0) + 1
    out["first_tunnel_goal"] = (tunnel_goals[0].get("action"), (tunnel_goals[0].get("extra") or {}).get("key"),
                                str(tunnel_goals[0].get("reason"))[:200]) if tunnel_goals else None
    battles = [r for r in records if isinstance(r.get("extra"), dict) and "battle" in r["extra"] and "kind" in r["extra"]]
    trainer_in_tunnel = [r for r in battles if r["extra"].get("kind") == "trainer" and in_tunnel(r)]
    out["tunnel_trainer_battles"] = [(r.get("frame"), r["extra"].get("began"), r["extra"].get("outcome")) for r in trainer_in_tunnel]
    if trainer_in_tunnel:
        began = trainer_in_tunnel[0]["extra"].get("began") or trainer_in_tunnel[0].get("frame")
        before = [r for r in goals if (r.get("frame") or 0) <= began]
        last = before[-1] if before else None
        out["goal_before_grunt_battle"] = (last.get("action"), (last.get("extra") or {}).get("key"),
                                           str(last.get("reason"))[:200], last.get("frame")) if last else None
    first_tunnel = out["first_tunnel_frame"]
    out["heal_known_fight_after_tunnel"] = sum(
        1 for r in goals if r.get("action") == "heal" and "a known fight is next" in str(r.get("reason"))
        and first_tunnel is not None and (r.get("frame") or 0) >= first_tunnel)
    out["milestones_told"] = [str(r.get("reason"))[:160] for r in records
                              if r.get("decision") == "tell" and "MILESTONE " in str(r.get("reason", ""))]
    return out


def control_stream(records: list[dict]) -> list:
    """The decision-record stream a CONTROL compares: decide, goal, goal_end, instructed and every press (frame, decision,
    action, reason, extra), the eyes' wall-clock `cost=` normalised."""
    keep = []
    for r in records:
        d, a = r.get("decision"), str(r.get("action", ""))
        if d in ("decide", "goal", "goal_end", "instructed") or a.startswith("press"):
            keep.append([r.get("frame"), d, a, re.sub(r"cost=\d+\.\d+ms", "cost=N", str(r.get("reason"))),
                         json.dumps(r.get("extra"), sort_keys=True, ensure_ascii=False)])
    return keep

# ------------------------------------------------------------------------------------------------ the orchestrator
def kill_family(pid: int) -> None:
    import psutil

    try:
        family = psutil.Process(pid).children(recursive=True) + [psutil.Process(pid)]
    except psutil.Error:
        return
    for p in family:
        try:
            p.kill()
        except psutil.Error:
            pass


def run_batch(batch: str) -> int:

            if proc.poll() is not None:

                running.remove(item)

        time.sleep(5)
    after = {arm: tree_digest(STAGE / f"arm{arm}" / "ai_player") for arm in ("A", "B")}

                       f"({sum(1 for r in results if not r.get('missing'))} of {len(results)} ran); read the evidence\n",
                       encoding="utf-8")
    subprocess.run([sys.executable, "-B", str(HERE / "ledger_tool.py"), "finish", str(entry), str(verdict),
                    "ai_player/evidence/2026-11-05-ch5-stops-development.txt"])

    return 0

def runs(batch: str) -> dict:
    out = {}
    for path in sorted((HERE / "out" / batch).glob("*.json")):
        if path.name.endswith((".job.json", ".stream.json")) or path.name == "results.json":
            continue
        r = json.loads(path.read_text(encoding="utf-8"))
        out[r["name"]] = r
    return out


def order(r: dict) -> list[str]:
    seq = []
    for _, action, reason in r["analysis"]["instructed"]:
        if action == "choose":
            seq.append(f"choose:{reason}")
        elif action == "back":
            seq.append(f"back:{reason.split(':')[0]}")
        else:
            seq.append(f"{action}:{reason[:40]}")
    return seq


def var(r: dict, which: str = "after") -> int | None:
    return r[f"marks_{which}"].get("RUSTBORO_CITY_STATE")


def frames(r: dict) -> int | None:
    return (r.get("harness_stop") or {}).get("real_frames")


def fresh(r: dict) -> bool:
    return all(c.get("status") == "PASS" for c in r["checks"] if isinstance(c, dict)) and bool(r["checks"])
