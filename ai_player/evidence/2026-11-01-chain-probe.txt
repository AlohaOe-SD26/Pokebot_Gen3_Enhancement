"""CHAIN PROBE (development, not graded): three UNBROKEN chains on today's code, each from a real New Game through chapter 1
(ch1_pokedex), 2 (ch2_wally), 3 (ch3_rustboro) and 4 (ch4_roxanne), one per personality and starter (Story/TREECKO,
Catcher/TORCHIC, Trainer/MUDKIP), each chain in its own disposable profile, the three in parallel. Each leg starts from the
state AND the world memory the previous leg left (live_step5's method, `run_arm` with a carried dict). Every leg goes through
the real launcher (purity per launch); what a leg reached is read from the GAME (truth) on the state it left, freshness-checked
(L5.THIS_RUN). DEVELOPMENT seeds 5983-5985 (one ledger entry; the prediction's sha256 is its bounds field). No PASS.

    py -3.12 chain_probe.py            the orchestrator (refuses unless HEAD is origin/main and the checkout is clean)
    py -3.12 chain_probe.py --child N  one chain (N = 1, 2, 3)
"""
import hashlib
import json
import os
import subprocess
import sys
import time
from pathlib import Path

REPO = Path("C:/Remote Projects/Pokebot_Gen3_Enhancement")
sys.path.insert(0, str(REPO))
HERE = Path(__file__).resolve().parent
CHAINS = {1: ("Story", "TREECKO"), 2: ("Catcher", "TORCHIC"), 3: ("Trainer", "MUDKIP")}
SEED0 = 5982
LEGS = [("L1", "ch1_pokedex", 330000), ("L2", "ch2_wally", 330000), ("L3", "ch3_rustboro", 600000),
        ("L4", "ch4_roxanne", 900000)]
WALL = 3600.0
PREDICTION = ("CHAIN PROBE prediction (registered by sha256 in the ledger entry before the runs): on today's code every chain "
              "reaches ch1_pokedex, ch2_wally and ch3_rustboro (the 2026-10-30 r1base arms reached chapters 1-2 in 5 of 6 "
              "graded arms; the 2026-10-31 held-out reached Rustboro in 6 of 6 from chapter-2 saves). Chapter 4 has never been "
              "played: the 2026-10-15 recon predicts the Rustboro gym's trainers and ROXANNE's level as blockers (MAX_LOSSES 3, "
              "no training after a loss), so ch4_roxanne is predicted NOT reached in at least 2 of 3 chains.")


def git(*args: str) -> str:
    return subprocess.run(["git", "-C", str(REPO), *args], capture_output=True, text=True, check=True).stdout.strip()


def child(n: int) -> None:
    from ai_player.gates import live_step5 as L5
    L5.PROFILE = f"ai-chain-probe-{n}"
    personality, starter = CHAINS[n]
    seed = SEED0 + n
    build_root = L5.launcher.resolve_build_root(None)
    carried: dict = {}
    legs_out = []
    previous = None
    for name, target, frames in LEGS:
        arm_name = f"C{n}{name}"
        env = L5.play_env(personality, starter if previous is None else None, target)
        arm = L5.Arm(arm_name, f"chain {n} ({personality}, {starter}) leg {name}: to {target}", frames, WALL, env, seed=seed,
                     fresh=previous is None, state_from=previous)
        run = L5.run_arm(arm, build_root, carried)
        L5.check_after_is_this_run(run)
        after = run.after or {}
        world = run.run_dir / "world.jsonl" if run.run_dir else None
        legs_out.append({
            "leg": name, "target": target, "frames_budget": frames, "exit": run.exit_code,
            "elapsed": round(run.elapsed or 0, 1), "run_dir": str(run.run_dir),
            "verdict": (run.verdict or {}).get("verdict"), "counters": (run.verdict or {}).get("counters"),
            "fresh": [getattr(c, "__dict__", str(c)) for c in run.checks][-1:],
            "map_after": after.get("map"), "marks_after": L5.marks(after),
            "milestones_told": [str(r.get("reason"))[:240] for r in L5.tells(run, "MILESTONE ")],
            "log_tail": run.ai_log.splitlines()[-14:], "harness_stop": run.measures.get("harness_stop"),
            "world_bytes": world.stat().st_size if world and world.is_file() else None, "notes": run.notes,
        })
        (HERE / f"chain-out-{n}.json").write_text(json.dumps({"n": n, "personality": personality, "starter": starter,
                                                              "seed": seed, "legs": legs_out}, indent=1, ensure_ascii=False,
                                                             default=str), encoding="utf-8")
        print(f"[C{n}] {name} exit {run.exit_code} {run.elapsed:.0f}s map {after.get('map')}", flush=True)
        if arm_name not in carried:
            print(f"[C{n}] {name} left no state and world memory to carry: the chain stops here", flush=True)
            break
        previous = arm_name


def summary_line(leg: dict) -> str:
    hs = leg.get("harness_stop") or {}
    end = next((line for line in reversed(leg.get("log_tail") or []) if "[AI-EXIT]" in line), "")
    return (f"  {leg['leg']} to {leg['target']}: exit {leg['exit']}, {leg['elapsed']} s, real frames {hs.get('real_frames')} "
            f"of {leg['frames_budget']}, purity {leg['verdict']}, end map {leg['map_after']}, world.jsonl {leg['world_bytes']} B"
            f"\n    told milestones: {' | '.join(leg['milestones_told'])[:900]}\n    {end[:400]}")


def orchestrate() -> int:
    from ai_player.gates import ledger
    from ai_player.gates import live_step5 as L5
    git("fetch", "-q", "origin")
    if git("rev-parse", "HEAD") != git("rev-parse", "origin/main"):
        raise SystemExit("REFUSED: HEAD is not origin/main")
    if git("status", "--porcelain"):
        raise SystemExit("REFUSED: the checkout is not clean")
    head = git("rev-parse", "--short", "HEAD")
    tree = git("rev-parse", "HEAD^{tree}")[:16]
    entry, lines = ledger.begin("story", "chain_probe", "development", SEED0 + 1, SEED0 + 3,
                                hashlib.sha256(PREDICTION.encode("utf-8")).hexdigest(), tree,
                                note=f"CHAIN PROBE: three unbroken New Game chains to ch4_roxanne on HEAD {head}; the "
                                     f"prediction's sha256 is the bounds field")
    print("\n".join(lines), flush=True)
    env = {**os.environ, "PYTHONUTF8": "1", "PYTHONDONTWRITEBYTECODE": "1"}
    procs = {}
    for n in CHAINS:
        log = open(HERE / f"chain-child-{n}.log", "w", encoding="utf-8")
        procs[n] = (subprocess.Popen([sys.executable, "-B", str(Path(__file__).resolve()), "--child", str(n)], cwd=str(REPO),
                                     env=env, stdout=log, stderr=subprocess.STDOUT), log)
        time.sleep(20)
    for n, (proc, log) in procs.items():
        proc.wait()
        log.close()
    results = []
    for n in CHAINS:
        path = HERE / f"chain-out-{n}.json"
        results.append(json.loads(path.read_text(encoding="utf-8")) if path.is_file() else {"n": n, "missing": True})
    ev = L5.EVIDENCE_DIR / f"{time.strftime('%Y-%m-%d')}-chain-probe.txt"
    out = [f"CHAIN PROBE - DEVELOPMENT, NOT GRADED, NO PASS: three UNBROKEN New Game chains on today's code, chapter 1 to "
           f"chapter 4 (ch1_pokedex, ch2_wally, ch3_rustboro, ch4_roxanne), one per personality and starter",
           f"date: {time.strftime('%Y-%m-%d %H:%M:%S')}   HEAD {head} (tree {tree})   seeds {SEED0 + 1}-{SEED0 + 3}, ledger "
           f"entry #{entry['id']} (development)",
           "setup: each chain in its own profile; each leg from the state and world memory the previous leg left (live_step5's "
           "run_arm); legs of 330,000 / 330,000 / 600,000 / 900,000 frames, 3,600 s each; the 12-lane bot tuner running",
           f"PREDICTION (sha256 in the ledger entry, before the runs): {PREDICTION}",
           "What the GAME says each leg reached, and how it ended:"]
    for r in results:
        out.append("")
        if r.get("missing"):
            out.append(f"CHAIN {r['n']}: NO RESULT FILE (see the child log in the session scratch)")
            continue
        out.append(f"CHAIN {r['n']}: {r['personality']}, {r['starter']}, seed {r['seed']}")
        for leg in r["legs"]:
            out.append(summary_line(leg))
            out.append(f"    marks after: {json.dumps(leg['marks_after'], ensure_ascii=False)[:700]}")
            out.append(f"    freshness: {json.dumps(leg['fresh'], ensure_ascii=False, default=str)[:500]}")
            out += [f"    log: {line[:300]}" for line in leg["log_tail"][-8:]]
    ev.write_text("\n".join(out) + "\n", encoding="utf-8")
    done = sum(1 for r in results if not r.get("missing"))
    ledger.finish(entry["id"], f"DEVELOPMENT, not graded: the chain probe ({done} of 3 chains ran; read the evidence)",
                  str(ev.relative_to(REPO)).replace("\\", "/"))
    print(f"[CHAIN] evidence: {ev}", flush=True)
    for r in results:
        if r.get("missing"):
            continue
        print(f"[C{r['n']}] " + "; ".join(f"{leg['leg']} {leg['target']} exit {leg['exit']} map {leg['map_after']}"
                                         for leg in r["legs"]), flush=True)
    return 0


if __name__ == "__main__":
    if len(sys.argv) == 3 and sys.argv[1] == "--child":
        child(int(sys.argv[2]))
    else:
        sys.exit(orchestrate())
