## 6. `MB_CRACKED_FLOOR` — exact mechanics (MEASURED, `src/field_tasks.c:801-842`)

```c
static void CrackedFloorPerStepCallback(u8 taskId)
{
    PlayerGetDestCoords(&x, &y);
    behavior = MapGridGetMetatileBehaviorAt(x, y);
    if (tFloor1Delay != 0 && (--tFloor1Delay) == 0) SetCrackedFloorHoleMetatile(tFloor1X, tFloor1Y);   // L810-811
    if (tFloor2Delay != 0 && (--tFloor2Delay) == 0) SetCrackedFloorHoleMetatile(tFloor2X, tFloor2Y);   // L812-813
    if (MetatileBehavior_IsCrackedFloorHole(behavior))
        VarSet(VAR_ICE_STEP_COUNT, 0);                        // L815-816  ← no speed guard
    if (x == tPrevX && y == tPrevY) return;                   // L819-820
    tPrevX = x; tPrevY = y;
    if (MetatileBehavior_IsCrackedFloor(behavior))            // L824

            VarSet(VAR_ICE_STEP_COUNT, 0);                    // L827  ← this is the fall

}
```

def _witness(report: dict, label: str, *, extra: dict | None = None) -> None:
    """Record a reading WITH ITS ORDER AND ITS POSITION. Called at the point the reading is taken.

    ⛔ A `readings` SET cannot express order, and order is exactly what separates "descended, then
    proved it can continue" from "proved it can continue, then descended" (the second is what a
    hoisted proof or a resume branch produces, and it is indistinguishable in a set).
    """
    trace: list = report.setdefault("trace", [])
    where = _where()
    trace.append({
        "seq": len(trace),
        "label": label,
        "map": (MapRSE(where[0]).name if isinstance(where, tuple) else str(where)),
        "tile": (tuple(where[1]) if isinstance(where, tuple) else None),
        "frame": context.emulator.get_frame_count(),
        **(extra or {}),
    })
    report["readings"].add(label)


def _arrive_and_check(map_enum, planned_tiles, report: dict) -> None:
    """On ARRIVAL at a floor: record the floor, pin the planner, and check the LIVE grid.

    ⛔ `_live_collision_grid` RAISES unless the avatar is standing on `map_enum`
    (sootopolis.py:885-889), so this is the ONLY moment the check can legally be taken.

    if not report["floors_seen"] or report["floors_seen"][-1] != map_enum.name:

    report["pin_readings"] = report.get("pin_readings", []) + [_pin_planner_to_live_layout(map_enum)]
    walls = _live_wall_tiles(map_enum, planned_tiles)
    report["live_checks_done"] = report.get("live_checks_done", 0) + 1

def _fall_through_crack(direction: str, *, from_map, crack_tile, expect_map,
                        report: dict, max_frames: int = 600) -> Generator:
    """The DELIBERATE FALL. ⛔ ITS SUCCESS PREDICATE IS THE **MAP**, NEVER THE COORDINATES.

    ⛔⛔ `_bounded_step` CANNOT BE USED HERE AND IT FAILS SILENTLY-THEN-LOUDLY IF YOU TRY.
    Its success test is `_at(map_enum, expect)` — SAME map AND exact tile (sootopolis.py:1068,
    :1095). After a CORRECT fall the avatar is on 3F, so `_at(4F, (6,4))` is False; it then presses
    the SAME direction twice more **on 3F**, walking the bot around inside the sealed pocket, and
    finally raises the Route-131-flavoured error at :1109. A correct run would fail, loudly, in the
    wrong words. ⇒ this is its own primitive.
    ⛔ AND A COORDINATE ASSERT IS SATISFIED BY NOT FALLING: `warphole` preserves (x,y) (RECON), so
    the avatar reads the crack tile before AND after. The map pair is the only witness that means
    anything. The ~80 frames of `lockall` + the fall animation are why max_frames is 600, not 120.
    """
    from modules.modes import BotModeError
    from modules.modes.util.walking import walk_one_tile
    from modules.story.early_game import _check_abort

    before_map, before_tile = _where()[0], _here_tile()
    stepper = walk_one_tile(direction, run=False)      # run=False is a SAFETY arg (sootopolis.py:1044)
    frames = 0
    try:
        while not _on_map(expect_map):
            _check_abort()                              # ⛔ loop head, OUTSIDE any try
            try:
                next(stepper)
            except StopIteration:
                pass                                    # the step finished; the fall script has not
            frames += 1
            if frames > max_frames:
                break
            yield
    finally:
        context.emulator.reset_held_buttons()           # ⛔ load-bearing (sootopolis.py:1073-1080)

    yield from wait_until_truly_controllable()
    if not _on_map(expect_map):
        raise BotModeError(
            f"Ch21 leg 4: stepped '{direction}' onto {crack_tile} expecting to FALL from "
            f"{from_map.name} to {expect_map.name}; after {frames} frames the run is at {_where()}. "
            "The map did NOT change — the cracked-floor mechanic did not fire. ⛔ Do not 'fix' this "
            "by asserting the coordinate: warphole preserves (x,y), so a coordinate check passes "
            "when nothing happened."
        )
    landing = _here_tile()
    report.setdefault("falls", []).append({
        "seq": len(report.get("trace", [])),
        "from_map": MapRSE(before_map).name, "from_tile": before_tile,
        "crack_tile": tuple(crack_tile), "to_map": expect_map.name,
        "landing": landing, "landing_preserved_xy": landing == tuple(crack_tile), "frames": frames,
    })
