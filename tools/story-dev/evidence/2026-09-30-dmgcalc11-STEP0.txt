===== m1_step0.py (the probe) =====
"""Design v2 step 11, F6: the Step-0 LIVE reads (observe-only; nothing in production changes).

Wraps two call sites BY THE NAMES THE CALLERS USE (both are from-imports):
  modules.battle_handler.handle_battle_action_selection   -> reads at selection ENTRY and at its EXIT;
  modules.modes._listeners.handle_battle                  -> a read at the battle's END.
Each read is PURE `read_symbol` (never `get_battle_state()`, KI-108). The wrappers add NO yield and press nothing:
`result = yield from inner(...)` passes every frame through. One line per read goes to OUT, stamped with the frame.

Run from tools/story-dev (the harness's own cwd), exactly as the seeded wrapper runs the harness:
  PYTHONUTF8=1 py -3.12 -u <this> saves/post_mtpyre.ss1 --driver storyresume --frames 90000 --log <log>
with env M1_STEP0_OUT=<path of the reads file>.
"""
import os
import random
import runpy
import sys
from pathlib import Path

random.seed(1)
REPO = Path(os.getcwd()).resolve().parents[1]
sys.path.insert(0, os.getcwd())                  # tools/story-dev, as `py -c` would have it
sys.path.insert(0, str(REPO / "sandbox-pokebot"))
OUT = Path(os.environ["M1_STEP0_OUT"])
_fh = OUT.open("w", encoding="utf-8")


def _raw_reads(tag: str) -> None:
    try:
        from modules.context import context
        from modules.memory import read_symbol, unpack_uint16, unpack_uint32

        frame = context.emulator.get_frame_count()
        turn = read_symbol("gBattleResults", offset=0x13, size=1)[0]
        outcome = read_symbol("gBattleOutcome", size=1)[0]
        bide = read_symbol("gBideDmg", size=0x10)
        bide4 = [int.from_bytes(bide[i * 4:i * 4 + 4], "little", signed=True) for i in range(4)]
        act = list(read_symbol("gChosenActionByBattler", size=4))
        mv = read_symbol("gChosenMoveByBattler", size=8)
        mv4 = [unpack_uint16(mv[i * 2:i * 2 + 2]) for i in range(4)]
        mons = read_symbol("gBattleMons", size=0x160)
        cnt = read_symbol("gBattlersCount", size=1)[0]
        flags = unpack_uint32(read_symbol("gBattleTypeFlags", size=4))
        # Flash Fire: gBattleResources -> flags (+0x04) -> u32 flags[battler] (a double dereference; EWRAM-guarded)
        res = unpack_uint32(read_symbol("gBattleResources", size=4))
        ff = "null"
        if 0x02000000 <= res < 0x02040000:
            fptr = unpack_uint32(context.emulator.read_bytes(res + 4, 4))
            ff = (f"{[unpack_uint32(context.emulator.read_bytes(fptr + 4 * b, 4)) for b in range(2)]}"
                  if 0x02000000 <= fptr < 0x02040000 else f"bad-flags-ptr:{fptr:08x}")
        parts = []
        for b in range(2):
            rec = mons[b * 0x58:(b + 1) * 0x58]
            parts.append(
                f"b{b}:sp={unpack_uint16(rec[0:2])} hp={unpack_uint16(rec[0x28:0x2A])}/{unpack_uint16(rec[0x2C:0x2E])}"
                f" pid={unpack_uint32(rec[0x48:0x4C]):08x} moves={[unpack_uint16(rec[0x0C + 2 * i:0x0E + 2 * i]) for i in range(4)]}"
                f" pp={list(rec[0x24:0x28])} st1={unpack_uint32(rec[0x4C:0x50]):x}")
    except Exception as error:  # noqa: BLE001 - a scratch probe: record the failure, never break the run
        _fh.write(f"{tag} READ-FAILED {error!r}\n")
        _fh.flush()
        return
    try:
        from modules.battle_state import get_main_battle_callback
        cb = get_main_battle_callback()
    except Exception as error:  # noqa: BLE001
        cb = f"?{error!r}"
    _fh.write(f"f{frame} {tag} turn={turn} outcome={outcome} bide={bide4} act={act} mv={mv4} count={cnt} "
              f"flags={flags:x} ff={ff} cb={cb} " + " | ".join(parts) + "\n")
    _fh.flush()


import modules.battle_handler as _bh  # noqa: E402
import modules.modes._listeners as _ls  # noqa: E402

_orig_sel = _bh.handle_battle_action_selection
_orig_battle = _ls.handle_battle


def _sel_wrapper(*args, **kwargs):
    _raw_reads("ENTRY")
    result = yield from _orig_sel(*args, **kwargs)
    _raw_reads("EXIT")
    return result


def _battle_wrapper(*args, **kwargs):
    _raw_reads("BATTLE-START")
    result = yield from _orig_battle(*args, **kwargs)
    _raw_reads("BATTLE-END")
    return result


_bh.handle_battle_action_selection = _sel_wrapper
_ls.handle_battle = _battle_wrapper
_fh.write("# m1_step0 armed: battle_handler.handle_battle_action_selection and _listeners.handle_battle wrapped\n")
_fh.flush()
sys.argv = ["story_harness.py"] + sys.argv[1:]
runpy.run_path("story_harness.py", run_name="__main__")

===== m1_step0_analyze.py (the analyser) =====
"""Analyse the Step-0 reads (design v2 step 11, F6). Each EXIT where our action is USE_MOVE (0) with a nonzero chosen
move is paired with the NEXT read (an ENTRY or a BATTLE-END), and the design's questions are answered from the numbers:
  Q1 on a clean non-KO hit, d(gBideDmg[1]) == the foe's HP difference;
  Q2 on a KO, d(gBideDmg[1]) >= the foe's HP before (the uncapped roll);
  Q3 the chosen move is final at EXIT (our chosen move's PP dropped by 1 when a turn ran);
  Q4 the turn counter at a battle-end read equals EXIT's turn (and is EXIT's + 1 at the next ENTRY);
  Q5 gBattleMons is readable at the battle's end (the foe's record is present);
  Q6 the Flash Fire dereference reads a sane value.
"""
import re
import sys
from pathlib import Path

LINE = re.compile(r"^f(\d+) (\S+) turn=(\d+) outcome=(\d+) bide=\[([^\]]*)\] act=\[([^\]]*)\] mv=\[([^\]]*)\] "
                  r"count=(\d+) flags=([0-9a-f]+) (?:ff=(\[[^\]]*\]|\S+) )?cb=(\S+) (.*)$")
MON = re.compile(r"b(\d):sp=(\d+) hp=(\d+)/(\d+) pid=([0-9a-f]+) moves=\[([^\]]*)\] pp=\[([^\]]*)\] st1=([0-9a-f]+)")


def ints(s):
    return [int(x) for x in s.split(",") if x.strip()]


def parse(path):
    out = []
    for line in Path(path).read_text(encoding="utf-8").splitlines():
        m = LINE.match(line)
        if not m:
            continue
        mons = {}
        for mm in MON.finditer(m.group(12)):
            mons[int(mm.group(1))] = dict(sp=int(mm.group(2)), hp=int(mm.group(3)), maxhp=int(mm.group(4)),
                                          pid=mm.group(5), moves=ints(mm.group(6)), pp=ints(mm.group(7)),
                                          st1=int(mm.group(8), 16))
        out.append(dict(f=int(m.group(1)), tag=m.group(2), turn=int(m.group(3)), outcome=int(m.group(4)),
                        bide=ints(m.group(5)), act=ints(m.group(6)), mv=ints(m.group(7)), count=int(m.group(8)),
                        flags=int(m.group(9), 16), ff=m.group(10), cb=m.group(11), mons=mons))
    return out


def main(path):
    rs = parse(path)
    print(f"{len(rs)} reads parsed; tags: " + ", ".join(f"{t}={sum(r['tag'] == t for r in rs)}"
                                                      for t in ("BATTLE-START", "ENTRY", "EXIT", "BATTLE-END")))
    print("ff values seen:", sorted({r["ff"] for r in rs}))
    rows = []
    for i, r in enumerate(rs):
        if r["tag"] != "EXIT" or r["act"][0] != 0 or r["mv"][0] == 0:
            continue
        nxt = next((x for x in rs[i + 1:] if x["tag"] in ("ENTRY", "BATTLE-END")), None)
        if nxt is None:
            continue
        me0, foe0 = r["mons"][0], r["mons"][1]
        me1, foe1 = nxt["mons"][0], nxt["mons"][1]
        double = bool(r["flags"] & 1)
        dbide = nxt["bide"][1] - r["bide"][1]
        same_foe = foe0["pid"] == foe1["pid"]
        dhp = foe0["hp"] - foe1["hp"] if same_foe else None
        slot = me0["moves"].index(r["mv"][0]) if r["mv"][0] in me0["moves"] else None
        pp_drop = (me0["pp"][slot] - me1["pp"][slot]) if (slot is not None and me0["pid"] == me1["pid"]) else None
        kind = ("KO-END" if nxt["tag"] == "BATTLE-END" and nxt["outcome"] else
                "SWITCH" if not same_foe else "TURN")
        rows.append(dict(f=r["f"], kind=kind, double=double, turn=(r["turn"], nxt["turn"]), outcome=nxt["outcome"],
                         mv=r["mv"][0], foe_mv=r["mv"][1], act_foe=r["act"][1], foe=foe0["sp"],
                         hp=(foe0["hp"], foe1["hp"] if same_foe else "-"), dbide=dbide, dhp=dhp, pp_drop=pp_drop,
                         foe_st=(foe0["st1"], foe1["st1"] if same_foe else "-")))
    print(f"\n{len(rows)} of our USE_MOVE exits, each paired with the next read:")
    print(" frame     kind    dbl turn(E->N) out  our_mv foe_act/mv foe_sp  hp(before->after)  d_bide  d_hp  pp_drop  foe_st1")
    for x in rows:
        print(f" f{x['f']:<8} {x['kind']:<7} {str(x['double'])[0]}  {x['turn'][0]:>3}->{x['turn'][1]:<3}   {x['outcome']}  "
              f"{x['mv']:>5}  {x['act_foe']}/{x['foe_mv']:<5}  {x['foe']:>5}   {str(x['hp']):<16} {x['dbide']:>6} "
              f"{str(x['dhp']):>5}   {str(x['pp_drop']):>4}    {x['foe_st']}")
    singles = [x for x in rows if not x["double"]]
    clean = [x for x in singles if x["kind"] == "TURN" and x["dhp"] is not None and x["dhp"] >= 0 and x["foe_st"][0] == 0
             and x["foe_st"][1] == 0]
    q1 = [(x["f"], x["dbide"], x["dhp"]) for x in clean if x["dbide"] != x["dhp"]]
    print(f"\nQ1 clean singles non-KO turns: {len(clean)}; d_bide == d_hp on {len(clean) - len(q1)}; disagreements: {q1}")
    kos = [x for x in singles if x["kind"] == "KO-END"]
    print(f"Q2 battle-end KO pairs: {len(kos)}; d_bide >= hp_before on "
          f"{sum(x['dbide'] >= x['hp'][0] for x in kos)}: " + str([(x['f'], x['hp'][0], x['dbide']) for x in kos]))
    sw = [x for x in singles if x["kind"] == "SWITCH"]
    print(f"   trainer-KO (foe identity changed) pairs: {len(sw)}: " + str([(x['f'], x['hp'][0], x['dbide']) for x in sw]))
    ran = [x for x in singles if x["pp_drop"] is not None]
    print(f"Q3 our chosen move's PP drop per pair: " + str(sorted({x['pp_drop'] for x in ran})) +
          f" (1 = the chosen move ran; 0 = it did not; n={len(ran)})")
    q4a = [(x["f"], x["turn"]) for x in singles if x["kind"] == "TURN" and x["turn"][1] != x["turn"][0] + 1]
    q4b = [(x["f"], x["turn"]) for x in kos if x["turn"][1] != x["turn"][0]]
    print(f"Q4 ENTRY turn == EXIT + 1: violations {q4a}; battle-end turn == EXIT's: violations {q4b}")
    ends = [r for r in rs if r["tag"] == "BATTLE-END"]
    print(f"Q5 battle-end reads: {len(ends)}; foe record present (maxhp > 0) on {sum(r['mons'][1]['maxhp'] > 0 for r in ends)}")


if __name__ == "__main__":
    main(sys.argv[1])
