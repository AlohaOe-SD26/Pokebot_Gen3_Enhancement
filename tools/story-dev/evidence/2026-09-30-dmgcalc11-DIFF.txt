"""M1b-DIFF (design v2 step 11, F7 / the arms table), exact form:
(a) the new battle_action_selection.py and battle_handler.py differ from the base (241929d7) by EXACTLY the added
    `from modules import dmgaudit as _dmgaudit` import and the hook calls (compared as ASTs with those removed);
(b) with the flag UNSET, every hook performs ZERO memory reads and ZERO log calls;
(c) with the flag SET, the hooks make ZERO `get_battle_state()` calls (KI-108), and each hook's lines use the category
    DMGAUDIT.
"""
import ast
import os
import pathlib
import subprocess
import sys
import types

WT = pathlib.Path(r"C:/Remote Projects/pokebot-wt-ch24")
sys.path.insert(0, str(WT / "sandbox-pokebot"))
BASE = "241929d7"
OK = []


def check(label, ok, detail=""):
    OK.append(ok)
    print(f"  {'PASS' if ok else 'FAIL'}  {label}" + (f"  — {detail}" if detail else ""))


def strip_hooks(src: str) -> str:
    tree = ast.parse(src)

    class Strip(ast.NodeTransformer):
        def visit_ImportFrom(self, node):
            if node.module == "modules" and any(a.name == "dmgaudit" for a in node.names):
                return None
            return node

        def visit_Expr(self, node):
            c = node.value
            if (isinstance(c, ast.Call) and isinstance(c.func, ast.Attribute) and isinstance(c.func.value, ast.Name)
                    and c.func.value.id == "_dmgaudit"):
                return None
            return node

    return ast.dump(Strip().visit(tree))


print("(a) the new modules = the base + the hooks only")
for rel in ("sandbox-pokebot/modules/battle_action_selection.py", "sandbox-pokebot/modules/battle_handler.py"):
    old = subprocess.run(["git", "-C", str(WT), "show", f"{BASE}:{rel}"], capture_output=True, text=True,
                         encoding="utf-8", check=True).stdout
    new = (WT / rel).read_text(encoding="utf-8")
    hooks = sum(1 for n in ast.walk(ast.parse(new)) if isinstance(n, ast.Call) and isinstance(n.func, ast.Attribute)
                and isinstance(n.func.value, ast.Name) and n.func.value.id == "_dmgaudit")
    check(f"{rel.split('/')[-1]}: AST with the hooks removed == the base's", strip_hooks(new) == ast.dump(ast.parse(old)),
          f"{hooks} hook call(s)")
