"""Grade T1-4 (the 16-lane smoke run) or T1-3 (the live env-leak fault) from a finished x4_headless batch dir.

usage: py -3.12 t1_smoke_grade.py <batch_dir> [smoke|fault]
Every check computes its verdict from the batch's own files; an absent observation is reported ABSENT, never PASS.
"""
import json
import re
import statistics
import sys
from pathlib import Path

bd = Path(sys.argv[1])
mode = sys.argv[2] if len(sys.argv) > 2 else "smoke"
summ = json.loads((bd / "summary.json").read_text(encoding="utf-8"))
arms = json.loads((bd / "ARMS.json").read_text(encoding="utf-8"))
viol = summ.get("header_violations", [])
kinds = summ.get("violation_kinds", {})
detail = summ["lanes_detail"]
rows = []


def row(name, ok, info=""):
    rows.append((name, ok))
    print(f"  {'PASS' if ok else 'FAIL'}  {name}{('  — ' + info) if info else ''}")


print(f"batch {bd.name}: verdict {summ['verdict']!r}, wall {summ.get('wall_minutes')} min, kinds {kinds}")
print(f"arms {arms.get('arm_names')}, tag {arms.get('profile_tag')}, head {arms.get('head', '')[:12]}, "
      f"stripped {arms.get('stripped_inherited')}")

if mode == "fault":
    row("T1-3 the batch ABORTED on an ARM HEADER violation", "ARM HEADER" in summ["verdict"], summ["verdict"])
    arm_v = [v for v in viol if v.get("kind") == "arm"]
    row("T1-3 the violation names H1 and POKEBOT_X4H_LEAK",
        any(v["tag"] == "H1" and "POKEBOT_X4H_LEAK" in v["reason"] for v in arm_v), str(arm_v)[:300])
    row("T1-3 no OTHER lane had an arm violation", all(v["tag"] == "H1" for v in arm_v), str([v["tag"] for v in arm_v]))
    row("T1-3 ARMS.json exists and records the fault", arms.get("fault_envleak") is True)
    h1 = detail.get("H1", {})
    sp = h1.get("spawns", [])
    row("T1-3 H1 was stopped (its spawn has an end time)", bool(sp) and sp[-1].get("end") is not None, str(sp)[:200])
else:
    row("T1-4 the time bound ended it (verdict OK)", summ["verdict"] == "OK", summ["verdict"])
    row("T1-4 zero violations of any kind", not viol, str(viol)[:400])
    for tag, d in sorted(detail.items(), key=lambda kv: int(kv[0][1:])):
        if not d.get("arm_headers_checked"):
            row(f"T1-4 {tag} had at least one arm header checked", False, "ABSENT")
    row("T1-4 every lane had >= 1 arm header checked", all(d.get("arm_headers_checked", 0) >= 1 for d in detail.values()),
        str({t: d.get("arm_headers_checked") for t, d in detail.items()}))
    b = {t: d for t, d in detail.items() if d.get("arm") == "B"}
    a = {t: d for t, d in detail.items() if d.get("arm") == "A"}
    row("T1-4 8 lanes per arm", len(a) == 8 and len(b) == 8, f"A={len(a)} B={len(b)}")
    fs = {t: d.get("monitor", {}).get("first_seen", {}) for t, d in detail.items()}
    for sub in (" rare=", "[KEYBATTLE]"):
        missing = [t for t in b if sub not in fs[t]]
        row(f"T1-4 all 8 B lanes showed {sub!r}", not missing, f"missing on {missing}; seconds "
            f"{sorted(fs[t].get(sub) for t in b if sub in fs[t])}")
    run = "consume the HM/counter runway"
    n_run = sum(run in fs[t] for t in a)
    row("T1-4 at least 6 of 8 A lanes showed the runway text (explicit '0' is OFF, live)", n_run >= 6, f"{n_run}/8")
    n_top = sum("why=topoff" in fs[t] and "[KIT-BUY]" in fs[t] for t in b)
    row("T1-4 at least 6 of 8 B lanes showed why=topoff AND [KIT-BUY]", n_top >= 6, f"{n_top}/8")
    # throughput: the highest frame stamp per log, summed per lane, over the lane's wall time
    fps = {}
    wall_s = float(summ.get("wall_minutes") or 0) * 60
    for tag, d in detail.items():
        slot = int(tag[1:]) - 1
        frames = 0
        for log in sorted((bd / f"L{slot + 1:02d}" / "debug-logs").glob("*.log")):
            mx = 0
            for m in re.finditer(r"^f(\d+) ", log.read_text(encoding="utf-8", errors="replace"), re.MULTILINE):
                mx = max(mx, int(m.group(1)))
            frames += mx
        lane_s = wall_s - slot * 6.0
        fps[tag] = round(frames / lane_s) if lane_s > 0 else 0
    med = statistics.median(fps.values()) if fps else 0
    row("T1-4 median frames/s per lane in [550, 700] (a THROUGHPUT finding, not a stop)", 550 <= med <= 700,
        f"median {med}; per lane {dict(sorted(fps.items(), key=lambda kv: int(kv[0][1:])))}")
    row("T1-4 the freeze watch never failed to answer", summ.get("freeze_unknown_checks", 0) == 0,
        str(summ.get("freeze_unknown_checks")))

fails = [n for n, ok in rows if not ok]
print(f"\n{len(rows) - len(fails)} of {len(rows)} PASS" + (f"; FAILED: {fails}" if fails else ""))
