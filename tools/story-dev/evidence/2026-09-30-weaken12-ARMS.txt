def stamped(path):
    return [l.rstrip("\n") for l in path.open(encoding="utf-8", errors="replace") if FR.match(l)]


def plain(lines):
    return [l for l in lines if "WEAKEN" not in l and "DMGAUDIT" not in l]


def sha(lines):
    return hashlib.sha256("\n".join(lines).encode()).hexdigest()[:16]


def frame(l):
    return int(FR.match(l).group(1))


def first_diff(a, b):
    for i, (x, y) in enumerate(zip(a, b)):
        if x != y:
            return i
    return None if len(a) == len(b) else min(len(a), len(b))


def befores(lines):
    out = {}
    for i, l in enumerate(lines):
        if "[DMGAUDIT] BEFORE" in l:
            r = adc.parse_line(l, i)
            out[r.frame] = r
    return out


def pairs(lines):
    rep = adc.audit_text("\n".join(lines))
    return {v.before.frame: v for v in rep.verdicts}


def catch_spans(lines):
    spans, cur = [], None
    for l in lines:
        if "[CATCH] start" in l:
            cur = [frame(l), None, re.search(r"kind=(\S+)", l).group(1)]
        elif "[CATCH] end" in l and cur is not None:
            cur[1] = frame(l)
            spans.append(tuple(cur))
            cur = None
    return spans


def weaken(lines, tag="[WEAKEN-OBS]"):
    return [l for l in lines if "WEAKEN" in l and (tag in l if tag else True)]
