import hashlib
import re
import sys
from pathlib import Path

FR = re.compile(r"^f(\d+) ")


def stamped(tag):
    p = O / f"{tag}.debug.txt"
    if not p.exists():
        return None
    return [l.rstrip("\n") for l in p.open(encoding="utf-8", errors="replace") if FR.match(l)]


def catches(lines):
    out, cur = [], None
    for l in lines:
        if "[CATCH] start" in l:
            cur = {"start": l, "turns": [], "end": None}
            out.append(cur)
        elif cur is not None and "[TURN" in l and "[BATTLE]" in l:
            cur["turns"].append(l)
        elif cur is not None and "[CATCH] end" in l:
            cur["end"] = l
            cur = None
    return out


for lever in ("NL", "ST", "DR"):
    r1, r2 = stamped(f"{lever}-r1"), stamped(f"{lever}-r2")
    print(f"===== {lever} =====")

        mv = lambda m: [(get_move_by_index(x).name, p) for x, p in zip(m.moves, m.pp) if x]
        print(f"{tag} BEFORE f{near} (turn line f{f}): act={r.act} mv={r.mv} bide={r.bide} weather={r.weather} "
              f"side={r.side} flags={r.flags:x}")
        print(f"   ours sp={me.species} L{me.level} hp={me.hp}/{me.maxhp} ab={me.ability} item={me.item} "
              f"st1={me.status1:x} moves={mv(me)}")
        print(f"   foe  sp={foe.species} L{foe.level} hp={foe.hp}/{foe.maxhp} ab={foe.ability} item={foe.item} "
              f"st1={foe.status1:x} st2={foe.status2:x} moves={mv(foe)} pp_total={sum(foe.pp)}")
    lines = [l for l in text.splitlines() if re.match(r"^f\d+ ", l)]
    print(f"{tag}: trainer battles {sum('[TRAINER] start' in l for l in lines)}, "
          f"catch battles {sum('[CATCH] start' in l for l in lines)}, "
          f"battles {sum('[BATTLE] started' in l for l in lines)}, shiny {sum('shiny' in l.lower() for l in lines if '[CATCH] start' in l)}")
