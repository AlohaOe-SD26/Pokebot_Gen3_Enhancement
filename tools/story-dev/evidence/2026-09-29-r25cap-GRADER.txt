# ---- r25_grade.py (the tracked-line extractor) ----
"""design v2 step 8 (RC) grader: extract the tracked lines from a Story-debug log, and compare two logs.

Tracked: [DEXFENCE], [DEXCATCH], [CATCH], [BATTLE] started/ended, [MAP] (minus the one-frame (0,0) and (255,255) warp reads), WHITEOUT.
Usage: r25_grade.py extract <debug.log> [max_frame]        -> prints the tracked lines
       r25_grade.py compare <a.log> <b.log> [max_frame]     -> first difference, or IDENTICAL with the count
"""
import re
import sys

KEEP = re.compile(r"\[(DEXFENCE|DEXCATCH|CATCH|MAP|KEYBATTLE)\]|\[BATTLE\] (started|ended)|WHITEOUT")
FRAME = re.compile(r"^f(\d+) ")


def tracked(path, max_frame=None):
    out = []
    with open(path, encoding="utf-8", errors="replace") as fh:
        for line in fh:
            line = line.rstrip("\n")
            m = FRAME.match(line)
            if not m or not KEEP.search(line):
                continue
            if max_frame is not None and int(m.group(1)) > max_frame:
                break
            if "[MAP]" in line and ("(0,0)" in line or "(255,255)" in line):
                continue
            out.append(line)
    return out


def main(argv):
    cmd = argv[1]
    if cmd == "extract":
        mf = int(argv[3]) if len(argv) > 3 else None
        for line in tracked(argv[2], mf):
            print(line)
        return 0
    a, b = argv[2], argv[3]
    mf = int(argv[4]) if len(argv) > 4 else None
    la, lb = tracked(a, mf), tracked(b, mf)
    for i, (x, y) in enumerate(zip(la, lb)):
        if x != y:
            print(f"FIRST DIFF at #{i} ({len(la)} vs {len(lb)} lines)\n    A: {x}\n    B: {y}")
            return 1
    if len(la) != len(lb):
        print(f"PREFIX IDENTICAL, lengths differ: {len(la)} vs {len(lb)}")
        return 1
    print(f"IDENTICAL: {len(la)} of {len(la)} tracked lines" + (f" (<= f{mf})" if mf else ""))
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv))
