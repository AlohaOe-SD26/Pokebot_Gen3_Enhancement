def seg(path):
    keep = set(tracked(str(path)))
    out, last = [], -1
    for line in open(path, encoding="utf-8", errors="replace"):

        f = int(m.group(1))
        if f < last:

        last = f
        if "[EARLY_BASKET]" in line or line in keep:

    return out
