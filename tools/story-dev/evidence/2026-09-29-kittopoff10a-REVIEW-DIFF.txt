    def nav(dest_map, dest_tile, **_kw):
        if back_ok or tuple(dest_map.value) != (0, 3):
            state["map"], state["tile"] = tuple(dest_map.value), tuple(dest_tile)
        else:
            state["map"], state["tile"] = tuple(dest_map.value), (1, 1)
        yield
    def one(*_a, **_kw):
        yield
    def face(d):
        state["facing"] = d; yield
    def press(b):
        if b == "A" and opens: state["menu"] = True
    def buy(ls):
        for item, qty in ls:
            state["asked"].append((item.name, qty))
            got = qty - (short if item.name == short_item else 0)
            state["bag"][item.name] = state["bag"].get(item.name, 0) + got
            if item.name == "Poké Ball" and qty >= 10:
                state["bag"]["Premier Ball"] = state["bag"].get("Premier Ball", 0) + 1
        yield
    class Bag:
        def quantity_of(self, item): return state["bag"].get(item.name, 0)
